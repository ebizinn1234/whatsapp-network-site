# COMANDOS PARA LIMPAR TODAS AS SESSÕES NO SERVIDOR

# 1. Parar o servidor
pm2 stop whatsapp-network

# 2. Limpar arquivos de sessão
rm -rf auth_info_*
rm -rf session_vault/*
rm -rf session_backups/*

# 3. Limpar banco de dados (execute no MySQL)
mysql -u root -p
USE whatsapp_network;
UPDATE whatsapp_sessions SET is_active = 0;
DELETE FROM whatsapp_sessions WHERE created_at < DATE_SUB(NOW(), INTERVAL 1 DAY);
EXIT;

# 4. Limpar logs
pm2 flush

# 5. Reiniciar servidor
pm2 start whatsapp-network

# 6. Verificar se funcionou
pm2 logs whatsapp-network --lines 20
